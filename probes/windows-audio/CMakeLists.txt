cmake_minimum_required(VERSION 3.24)
project(ScreenPilotWindowsAudioProbe LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Windows only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(screenpilot_windows_audio_probe main.cpp)
set_target_properties(screenpilot_windows_audio_probe PROPERTIES WIN32_EXECUTABLE FALSE)
target_compile_definitions(screenpilot_windows_audio_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(screenpilot_windows_audio_probe PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets d3d11 dxgi windowsapp user32 ole32 uuid avrt)
