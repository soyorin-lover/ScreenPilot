cmake_minimum_required(VERSION 3.24)
project(ScreenPilotWindowCaptureProbe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This probe requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Multimedia)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(screenpilot_window_capture_probe main.cpp)
set_target_properties(screenpilot_window_capture_probe PROPERTIES WIN32_EXECUTABLE FALSE)
target_link_libraries(screenpilot_window_capture_probe PRIVATE
    Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Multimedia
)
